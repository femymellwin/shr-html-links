EXPERIMENT 01 Manual pp. 1-5

Implementing Perceptron

One artificial neuron learns a linearly separable truth table.

Foundations10-15 min
AIM

Implement a perceptron learning algorithm for the AND gate, display its predictions, and draw the decision boundary.

Step by step

Procedure / algorithm

01

Prepare truth-table inputs

Define X as the four binary input pairs and T as expected AND outputs [0, 0, 0, 1].

02

Choose starting parameters

Use weights=[0.3, -0.1], bias=0.2, alpha=0.1 and a maximum of 10 epochs, matching the manual's actual code.

03

Calculate prediction

For each row calculate z = w1*x1 + w2*x2 + bias. Output 1 when z >= 0, otherwise 0.

04

Correct mistakes

If prediction differs from the target, update w = w + alpha*(target-output)*x and bias = bias + alpha*(target-output).

05

Stop when correct

Repeat complete passes (epochs) until all four predictions are correct or the epoch limit is reached.

06

Print and plot

Print the final weights, bias and truth table. Plot the four points and the line w1*x1 + w2*x2 + bias = 0.

Inputs & architecture

x1x2AND target
000
010
100
111

Quick revision

Memory hook

AND needs both inputs to be 1. Only (1,1) belongs to class 1.

Why a line works

The AND classes are linearly separable; a straight boundary can separate them.

Key terms

weight, bias, learning rate, activation, epoch, decision boundary

Exam-safe correction / important note
The algorithm text asks for zero initialization, while the printed program starts with nonzero weights and bias. This page follows the printed program to match its reported output.

# %% Part A - data and training
import numpy as np
import matplotlib.pyplot as plt

X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
T = np.array([0, 0, 0, 1])
weights = np.array([0.3, -0.1], dtype=float)
bias = 0.2
alpha = 0.1
epochs = 10

for epoch in range(epochs):
    errors = 0
    print(f"\nEpoch {epoch + 1}")
    for x_i, target in zip(X, T):
        z = np.dot(weights, x_i) + bias
        output = int(z >= 0)
        error = int(target) - output
        if error != 0:
            weights += alpha * error * x_i
            bias += alpha * error
            errors += 1
        print(f"X={x_i}, target={target}, output={output}, "
              f"weights={weights}, bias={bias}")
    if errors == 0:
        print("Training converged. Stopping early.")
        break
Cell 2: Part B - verification and boundary28 lines
# %% Part B - verification and boundary
predictions = (X @ weights + bias >= 0).astype(int)
print("\nFinal weights:", weights, "Final bias:", bias)
for x_i, prediction in zip(X, predictions):
    print(f"X={x_i}, prediction={prediction}")
print("All AND outputs correct:", np.array_equal(predictions, T))

plt.figure(figsize=(6, 5))
for target in (0, 1):
    points = X[T == target]
    plt.scatter(points[:, 0], points[:, 1],
                label=f"Target {target}")
# Added guard: do not divide by a zero second weight.
if not np.isclose(weights[1], 0):
    x1 = np.array([-0.1, 1.1])
    x2 = -(weights[0] * x1 + bias) / weights[1]
    plt.plot(x1, x2, label="Decision boundary")
elif not np.isclose(weights[0], 0):
    plt.axvline(-bias / weights[0], label="Decision boundary")
else:
    print("No nontrivial boundary: both weights are zero.")
plt.xlabel("x1")
plt.ylabel("x2")
plt.title("Perceptron Decision Boundary - AND")
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()

# %% Part A - model and manual weights
import numpy as np
import tensorflow as tf
import matplotlib.pyplot as plt
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Input, Dense
from tensorflow.keras.optimizers import Adam

X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]], dtype="float32")
y = np.array([[0], [1], [1], [0]], dtype="float32")
model = Sequential([
    Input(shape=(2,)),
    Dense(2, activation="sigmoid", name="hidden"),
    Dense(1, activation="sigmoid", name="output"),
])
model.compile(optimizer=Adam(learning_rate=1.0),
              loss="binary_crossentropy", metrics=["accuracy"])

W1 = np.array([[2.524, -5.875], [-2.715, 5.026]])
b1 = np.array([-1.429, -3.095])
W2 = np.array([[1.531], [3.240]])
b2 = np.array([-1.099])
model.get_layer("hidden").set_weights([W1, b1])
model.get_layer("output").set_weights([W2, b2])

print("Initial weights and biases:")
for layer in model.layers:
    print(layer.name, layer.get_weights())
initial_preds = model.predict(X, verbose=0).ravel()
print("Initial probabilities:", np.round(initial_preds, 4))
print("Initial classes:", (initial_preds > 0.5).astype(int))
# The manual checks the stopping condition AFTER each training epoch.
Cell 2: Part B - callback, training, evaluation and plots41 lines
# %% Part B - callback, training, evaluation and plots
class EpochLogger(tf.keras.callbacks.Callback):
    def on_epoch_end(self, epoch, logs=None):
        probs = self.model.predict(X, verbose=0).ravel()
        classes = (probs > 0.5).astype(int)
        print(f"\nEpoch {epoch + 1}")
        for row, prob, cls in zip(X, probs, classes):
            print(f"Input={row}, output={prob:.4f}, class={cls}")
        for layer in self.model.layers:
            w, b = layer.get_weights()
            print(layer.name, "weights:", np.round(w, 3))
            print("biases:", np.round(b, 3))
        if np.array_equal(classes, y.ravel().astype(int)):
            print("All predictions correct. Stopping early.")
            self.model.stop_training = True

history = model.fit(
    X, y, epochs=100, batch_size=4, verbose=0,
    validation_data=(X, y), callbacks=[EpochLogger()]
)
final_probs = model.predict(X, verbose=0).ravel()
print("\nFinal XOR predictions:")
for row, prob in zip(X, final_probs):
    print(f"Input={row}, output={prob:.4f}, class={int(prob > 0.5)}")
loss, accuracy = model.evaluate(X, y, verbose=0)
print(f"Final truth-table accuracy: {accuracy:.4f}; loss: {loss:.4f}")

# Separate figures contain the same curves as the manual.
for metric, title in [("accuracy", "Accuracy"), ("loss", "Loss")]:
    plt.figure(figsize=(6, 4))
    epoch_axis = np.arange(1, len(history.history[metric]) + 1)
    plt.plot(epoch_axis, history.history[metric], label="Train")
    plt.plot(epoch_axis, history.history["val_" + metric],
             linestyle="--", label="Validation (same four rows)")
    plt.xlabel("Epoch")
    plt.ylabel(title)
    plt.title(f"XOR - {title} over epochs")
    plt.grid(True)
    plt.legend()
    plt.tight_layout()
    plt.show()

import string

)
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.utils import to_categorical

# CHANGE ROOT to the folder containing english.csv and Img/.
ROOT = Path("data/CharacterR")
CSV_FILE = ROOT / "english.csv"
LABEL_MODE = "letters"  # "all" ONLY for an approved mixed-label task.
IMG_SIZE = 28
if not CSV_FILE.is_file():
    raise FileNotFoundError(f"Cannot find {CSV_FILE.resolve()}")
df = pd.read_csv(CSV_FILE, dtype=str)
if not {"image", "label"}.issubset(df.columns):
    raise ValueError("CSV must contain image and label columns.")
if df.empty or df[["image", "label"]].isna().any().any():
    raise ValueError("CSV is empty or has missing image/label values.")
df["label"] = df["label"].str.strip().str.upper()
labels = df["label"].to_numpy()
print(df.head())
print("Unique labels:", sorted(set(labels)))
if any(len(label) != 1 for label in labels):
    raise ValueError("Every label must be one character.")
if LABEL_MODE == "letters":
    class_names = list(string.ascii_uppercase)
    bad = sorted(set(labels) - set(class_names))
    if bad:
        raise ValueError(f"Non A-Z labels: {bad}. Confirm task scope.")
elif LABEL_MODE == "all":
    class_names = sorted(set(labels))
else:

# %% Part C - source CNN, training and test evaluation
model = Sequential([
    Input(shape=(IMG_SIZE, IMG_SIZE, 1)),
    Conv2D(32, (3, 3), activation="relu"),
    MaxPooling2D((2, 2)),
    Conv2D(64, (3, 3), activation="relu"),
    MaxPooling2D((2, 2)),
    Flatten(),
    Dense(128, activation="relu"),
    Dropout(0.5),
    Dense(num_classes, activation="softmax"),
])
model.compile(optimizer=Adam(), loss="categorical_crossentropy",
              metrics=["accuracy"])
model.summary()
history = model.fit(
    X_train, y_train, epochs=10, batch_size=64,
    validation_split=0.2, verbose=1
)
test_loss, test_acc = model.evaluate(X_test, y_test, verbose=0)
print(f"Test Accuracy: {test_acc * 100:.2f}%")
print(f"Test Loss: {test_loss:.4f}")

Oral examination

Viva questions and answers

Expand each question and explain it in your own words.

What does Conv2D do?+

Applies learned filters to image neighborhoods to produce feature maps.

What is pooling?+

A downsampling operation; max pooling keeps the maximum response within a local window.

Why use dropout?+

It randomly suppresses some activations during training to reduce reliance on individual neurons.

Why do output units equal the number of classes?+

Each softmax unit represents exactly one class under the chosen label mapping.

Why is ord(label)-ord("A") unsafe for digits?+

For a digit such as 0 it creates a negative number rather than a valid A-Z class index.

Why is the dataset needed before running?+

The program must read every actual image referred to in english.csv; the PDF includes example screenshots, not the image files.
